# The paths of OpenCCU-Base that openccu-lite builds from, for git filter-repo --paths-from-file.
# A commit that touched none of them is filtered out. Changing this list rewrites the whole
# history: the sync then refuses to push (see README.md), so a change is a new generation.
# It was changed twice, both times before anything was built from the old one: on 2026-10-07 and
# on 2026-10-08 (the WebUI's files addons read). This is the final list.

# the CMake build: the top-level list, the version header generator, the device type stripper
CMakeLists.txt
cmake/
build-tools/

# the licences (HMSL 2.0, GPL 2.0, LGPL 2.1) and their index
licenses/

# every component under src/ - daemons, libraries, Tcl modules, kernel modules, device types,
# tcl_homematic - except the WebUI and the daemons openccu-lite does not ship: hss_led (occulited
# drives the status LED), eq3configd (occulited answers the discovery), ssdpd (occulited
# announces the system)
regex:^src/(?!(webui|hss_led|eq3configd|ssdpd)/)

# the startup scripts the image installs to /bin
bin/hm_autoconf
bin/hm_deldev
bin/hm_startup

# installed as they are: /etc, /firmware
etc/
firmware/

# hmipserver (HMIPServer.jar with the ESHBridge on its classpath), its group definitions, and the
# HmIP coprocessor updater with its data. Not HMServer.jar (the BidCos-only server openccu-lite
# never starts), nor the WebUI's FreeMarker pages (openccu-lite brings the four group pages
# occulited reads) or the diagrams' measurement templates
opt/HMServer/HMIPServer.jar
opt/HMServer/coupling/
opt/HMServer/groups/
opt/HmIP/

# translations and its stringtable. Nothing else of the WebUI.
src/webui/www/config/img/devices/

src/webui/www_source/create_devdb_tcl.tcl
src/webui/www_source/utf82ansi.py

src/webui/www/config/stringtable_de.txt
